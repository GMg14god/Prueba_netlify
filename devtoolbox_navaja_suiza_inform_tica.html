Comandos Esenciales de Git

# Clonar repositorio existente

git clone https://github.com/user/repo.git

# Subir cambios paso a paso

git add .
git commit -m "Mensaje"
git push origin main

# Descargar cambios y sincronizar

git pull origin main

# Guardar cambios sin commitear temporalmente

git stash
git stash pop

# Deshacer último commit sin perder archivos

git reset --soft HEAD~1

Comandos Diarios de Linux & Servidores

# Ver puertos y servicios escuchando

sudo ss -tulpn

# Ver uso de disco en formato legible

df -h

# Buscar ficheros por nombre

find . -name "*.log" -type f

# Monitorizar logs en tiempo real

tail -f /var/log/syslog

# Comprimir y empaquetar una carpeta

tar -czvf respaldo.tar.gz /carpeta
